4. Auto-Calculate the workScore (0–100 Badge on Your Cards)

Instead of guessing the 78 or 85 badge shown on your spot cards, compute it automatically so cafes compete to improve their score:

```typescript
export function calculateWorkScore(spot: WorkSpot): number {
  let score = 0;

  // 1. WiFi Speed (up to 40 pts)
  const mbps = spot.workMetrics.wifi.downloadMbps;
  if (mbps >= 50) score += 40;
  else if (mbps >= 25) score += 30;
  else if (mbps >= 10) score += 20;
  else score += 10;

  // 2. Power Outlets (up to 25 pts)
  const outletMap = {
    "every-table": 25,
    "wall-seats-only": 15,
    scarce: 5,
    none: 0,
  };
  score += outletMap[spot.workMetrics.powerOutlets];

  // 3. Noise & Focus (up to 20 pts)
  const noiseMap = {
    silent: 20,
    "quiet-lo-fi": 20,
    "moderate-buzz": 12,
    loud: 4,
  };
  score += noiseMap[spot.workMetrics.noiseLevel];

  // 4. Value & Affordability (up to 15 pts)
  if (spot.pricing.entryType === "free-entry") score += 15;
  else if ((spot.pricing.americanoPriceRwf ?? 9999) <= 3000) score += 15;
  else score += 8;

  return Math.min(100, score);
}
```

```tsx
import { WorkSpot } from "@/types/spot";

interface SpotJsonLdProps {
  spot: WorkSpot;
  baseUrl?: string; // e.g., "https://workfromkigali.xyz"
}

export function SpotJsonLd({
  spot,
  baseUrl = "https://workfromkigali.xyz",
}: SpotJsonLdProps) {
  // Convert 0-100 WorkScore into a 1.0 - 5.0 star rating for Google Rich Snippets
  const normalizedRating = Number(
    Math.max(1, Math.min(5, (spot.workMetrics.workScore / 100) * 5)).toFixed(1),
  );

  // Map venue category to Schema.org types
  const schemaTypeMap: Record<WorkSpot["category"], string> = {
    cafe: "CafeOrCoffeeShop",
    coworking: "CoworkingSpace",
    library: "Library",
    "hotel-lounge": "Hotel",
  };

  // Parse "07:30 - 21:00" into Schema.org openingHoursSpecification
  const parseHours = (days: string[], range: string) => {
    const [opens, closes] = range.split("-").map((t) => t.trim());
    if (!opens || !closes) return null;
    return {
      "@type": "OpeningHoursSpecification",
      dayOfWeek: days,
      opens,
      closes,
    };
  };

  const openingHoursSpecification = [
    parseHours(
      ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"],
      spot.hours.weekdays,
    ),
    parseHours(["Saturday"], spot.hours.saturday),
    parseHours(["Sunday"], spot.hours.sunday),
  ].filter(Boolean);

  // Build Google-recognized amenities (WiFi, Power Outlets, A/C)
  const amenityFeature = [
    {
      "@type": "LocationFeatureSpecification",
      name: "Wi-Fi",
      value: true,
    },
    {
      "@type": "LocationFeatureSpecification",
      name: `Verified ${spot.workMetrics.wifi.downloadMbps} Mbps Internet`,
      value: true,
    },
    {
      "@type": "LocationFeatureSpecification",
      name: "Power Outlets for Laptops",
      value: spot.workMetrics.powerOutlets !== "none",
    },
    {
      "@type": "LocationFeatureSpecification",
      name: "Air Conditioning / Ventilation",
      value: spot.workMetrics.acOrVentilation,
    },
  ];

  // Format payment methods (Crucial for local Rwandan searchers)
  const paymentAccepted = [
    spot.pricing.acceptsMomo ? "MTN Mobile Money, Airtel Money" : null,
    spot.pricing.acceptsCard ? "Credit Card, Debit Card" : null,
    "Cash (RWF)",
  ]
    .filter(Boolean)
    .join(", ");

  const jsonLd = {
    "@context": "https://schema.org",
    "@type": schemaTypeMap[spot.category] || "CafeOrCoffeeShop",
    "@id": `${baseUrl}/spots/${spot.slug}#venue`,
    name: spot.name,
    description: spot.seo.metaDescription || spot.tagline,
    url: `${baseUrl}/spots/${spot.slug}`,
    image: [spot.media.coverImage, ...spot.media.gallery].filter(Boolean),
    priceRange: spot.pricing.americanoPriceRwf
      ? `~${spot.pricing.americanoPriceRwf.toLocaleString()} RWF`
      : "Free Entry",
    currenciesAccepted: "RWF",
    paymentAccepted,
    address: {
      "@type": "PostalAddress",
      streetAddress: spot.location.address,
      addressLocality: spot.location.neighborhood,
      addressRegion: "Kigali",
      addressCountry: "RW",
    },
    geo: {
      "@type": "GeoCoordinates",
      latitude: spot.location.coordinates.lat,
      longitude: spot.location.coordinates.lng,
    },
    hasMap: spot.location.googleMapsUrl,
    openingHoursSpecification,
    amenityFeature,
    // Workfrom Editorial Review & Score -> Triggers Google Star Rating Snippet
    review: {
      "@type": "Review",
      author: {
        "@type": "Organization",
        name: "Workfrom Kigali",
        url: baseUrl,
      },
      datePublished: spot.workMetrics.wifi.lastTestedAt,
      reviewBody: `${spot.name} in ${spot.location.neighborhood} scored ${spot.workMetrics.workScore}/100 for remote work, featuring verified ${spot.workMetrics.wifi.downloadMbps} Mbps WiFi and ${spot.workMetrics.powerOutlets.replace(/-/g, " ")} power outlets.`,
      reviewRating: {
        "@type": "Rating",
        ratingValue: normalizedRating,
        bestRating: "5",
        worstRating: "1",
      },
    },
  };

  return (
    <script
      type="application/ld+json"
      dangerouslySetInnerHTML={{
        // Escape '<' to prevent XSS injection from user-submitted spot names
        __html: JSON.stringify(jsonLd).replace(/</g, "\\u003c"),
      }}
    />
  );
}
```

4. The Weighted workScore Engine (lib/scoring.ts)

Put this in `lib/scoring.ts` so both your seed script and your frontend UI (e.g., if users submit live speed check-ins later) share the exact same scoring math.

```typescript
import { NoiseLevel, OutletAvailability, VenueCategory } from "@/types/spot";

export interface RawScoreInput {
  downloadMbps: number;
  uploadMbps?: number;
  stability: "rock-solid" | "good" | "spotty";
  timeLimitMinutes: number | null;
  powerOutlets: OutletAvailability;
  noiseLevel: NoiseLevel;
  seatingComfort: "ergonomic" | "standard-wood" | "outdoor-only";
  callFriendly: boolean;
  entryType: "free-with-purchase" | "free-entry" | "day-pass";
  americanoPriceRwf: number | null;
  dayPassPriceRwf: number | null;
  category: VenueCategory;
}

export interface ScoreBreakdown {
  total: number; // 0 - 100
  wifiPoints: number; // Max 40
  powerPoints: number; // Max 25
  focusPoints: number; // Max 20
  valuePoints: number; // Max 15
  gradeLabel:
    | "Elite Work Hub"
    | "Great for Deep Work"
    | "Solid Backup"
    | "Quick Stop Only";
}

export function calculateDetailedWorkScore(
  input: RawScoreInput,
): ScoreBreakdown {
  // 1. WIFI PERFORMANCE & RELIABILITY (Max 40 pts)
  let wifiPoints = 0;
  if (input.downloadMbps >= 60) wifiPoints = 32;
  else if (input.downloadMbps >= 35) wifiPoints = 26;
  else if (input.downloadMbps >= 15) wifiPoints = 19;
  else if (input.downloadMbps >= 5) wifiPoints = 12;
  else wifiPoints = 5;

  // Stability bonus/penalty
  if (input.stability === "rock-solid") wifiPoints += 8;
  else if (input.stability === "good") wifiPoints += 4;
  else wifiPoints -= 4;

  // Penalty if cafe kicks you off WiFi after 60/120 mins
  if (input.timeLimitMinutes !== null && input.timeLimitMinutes <= 120) {
    wifiPoints -= 6;
  }
  wifiPoints = Math.max(0, Math.min(40, wifiPoints));

  // 2. POWER OUTLETS & ERGONOMICS (Max 25 pts)
  const outletScores: Record<OutletAvailability, number> = {
    "every-table": 20,
    "wall-seats-only": 13,
    scarce: 5,
    none: 0,
  };
  let powerPoints = outletScores[input.powerOutlets];

  if (input.seatingComfort === "ergonomic") powerPoints += 5;
  else if (input.seatingComfort === "standard-wood") powerPoints += 3;
  powerPoints = Math.max(0, Math.min(25, powerPoints));

  // 3. FOCUS, NOISE & CALL FRIENDLINESS (Max 20 pts)
  const noiseScores: Record<NoiseLevel, number> = {
    silent: 16,
    "quiet-lo-fi": 17,
    "moderate-buzz": 11,
    loud: 4,
  };
  let focusPoints = noiseScores[input.noiseLevel];
  if (input.callFriendly) focusPoints += 3;
  focusPoints = Math.max(0, Math.min(20, focusPoints));

  // 4. VALUE & RWF AFFORDABILITY (Max 15 pts)
  let valuePoints = 8;
  if (input.entryType === "free-entry") {
    valuePoints = 15; // e.g., Kigali Public Library
  } else if (input.entryType === "free-with-purchase") {
    const coffee = input.americanoPriceRwf ?? 3000;
    if (coffee <= 2500) valuePoints = 15;
    else if (coffee <= 3500) valuePoints = 12;
    else valuePoints = 9;
  } else if (input.entryType === "day-pass") {
    const pass = input.dayPassPriceRwf ?? 15000;
    if (pass <= 10000) valuePoints = 13;
    else if (pass <= 20000) valuePoints = 10;
    else valuePoints = 7;
  }

  const total = Math.max(
    10,
    Math.min(
      100,
      Math.round(wifiPoints + powerPoints + focusPoints + valuePoints),
    ),
  );

  let gradeLabel: ScoreBreakdown["gradeLabel"] = "Solid Backup";
  if (total >= 85) gradeLabel = "Elite Work Hub";
  else if (total >= 75) gradeLabel = "Great for Deep Work";
  else if (total >= 60) gradeLabel = "Solid Backup";
  else gradeLabel = "Quick Stop Only";

  return {
    total,
    wifiPoints,
    powerPoints,
    focusPoints,
    valuePoints,
    gradeLabel,
  };
}
```

2. Automated `matchdayScore` Calculator (`lib/ruhago-scoring.ts`)

Mirrors `lib/scoring.ts` from Workfrom so every bar gets an objective 0–100 MatchdayScore™ based on what fans in Kigali actually care about:

```typescript
import { AudioPolicy, CrowdAtmosphere, ScreenSetup } from "@/types/venue";

export interface RawMatchdayInput {
  screenSetup: ScreenSetup;
  screenCount: number;
  multiMatchCapable: boolean;
  audio: AudioPolicy;
  atmosphere: CrowdAtmosphere;
  hasGeneratorBackup: boolean;
  streamSource: "dstv-satellite" | "canal-plus" | "fiber-iptv";
  openForLateKickoffs: boolean;
  entryType: "free-entry" | "minimum-spend" | "ticketed-big-match";
  localBeerPriceRwf: number | null;
}

export interface MatchdayScoreBreakdown {
  total: number; // 0 - 100
  screenPoints: number; // Max 35
  streamPoints: number; // Max 25 (Generator + DStv/Canal+ reliability)
  vibePoints: number; // Max 25 (Audio + Crowd energy + Late hours)
  valuePoints: number; // Max 15 (RWF drink affordability & free entry)
  gradeLabel:
    | "Iconic Matchday Arena"
    | "Prime Viewing Spot"
    | "Solid Neighborhood Bar"
    | "Casual Screen";
}

export function calculateMatchdayScore(
  input: RawMatchdayInput,
): MatchdayScoreBreakdown {
  // 1. SCREENS & VIEWING ANGLES (Max 35 pts)
  const setupScores: Record<ScreenSetup, number> = {
    "giant-projector": 25,
    "multiple-4k-tvs": 24,
    "single-tv": 12,
  };
  let screenPoints = setupScores[input.screenSetup];
  if (input.screenCount >= 4) screenPoints += 5;
  if (input.multiMatchCapable) screenPoints += 5;
  screenPoints = Math.max(0, Math.min(35, screenPoints));

  // 2. POWER & STREAM RELIABILITY (Max 25 pts — Huge Kigali differentiator)
  let streamPoints = 0;
  if (input.hasGeneratorBackup) streamPoints += 13;
  if (
    input.streamSource === "dstv-satellite" ||
    input.streamSource === "canal-plus"
  ) {
    streamPoints += 12; // Zero buffering vs IPTV
  } else {
    streamPoints += 6;
  }
  streamPoints = Math.max(0, Math.min(25, streamPoints));

  // 3. ATMOSPHERE, AUDIO & LATE KICKOFFS (Max 25 pts)
  const audioScores: Record<AudioPolicy, number> = {
    "full-commentary-loud": 12,
    "moderate-audio": 8,
    "muted-music": 2,
  };
  const vibeScores: Record<CrowdAtmosphere, number> = {
    buzzing: 9,
    lively: 8,
    relaxed: 6,
  };
  let vibePoints = audioScores[input.audio] + vibeScores[input.atmosphere];
  if (input.openForLateKickoffs) vibePoints += 4;
  vibePoints = Math.max(0, Math.min(25, vibePoints));

  // 4. VALUE & RWF AFFORDABILITY (Max 15 pts)
  let valuePoints = 8;
  const beer = input.localBeerPriceRwf ?? 2000;
  if (input.entryType === "free-entry" && beer <= 1800) valuePoints = 15;
  else if (input.entryType === "free-entry" && beer <= 2500) valuePoints = 12;
  else if (input.entryType === "free-entry") valuePoints = 9;
  else valuePoints = 7;

  const total = Math.max(
    10,
    Math.min(
      100,
      Math.round(screenPoints + streamPoints + vibePoints + valuePoints),
    ),
  );

  let gradeLabel: MatchdayScoreBreakdown["gradeLabel"] =
    "Solid Neighborhood Bar";
  if (total >= 86) gradeLabel = "Iconic Matchday Arena";
  else if (total >= 75) gradeLabel = "Prime Viewing Spot";
  else if (total >= 60) gradeLabel = "Solid Neighborhood Bar";
  else gradeLabel = "Casual Screen";

  return {
    total,
    screenPoints,
    streamPoints,
    vibePoints,
    valuePoints,
    gradeLabel,
  };
}
```

2. BroadcastEvent + SportsEvent JSON-LD Component (`components/seo/MatchBroadcastJsonLd.tsx`)

Google uses BroadcastEvent nested inside SportsEvent to understand which live game is being broadcast at which physical Kigali venue and at what kickoff time. Drop this onto your match detail pages (/matches/[slug]) and venue detail pages (/places/[slug]).

```tsx
import { MatchdayVenue } from "@/types/venue";

export interface MatchFixture {
  slug: string; // e.g., "apr-fc-vs-al-ahly-caf-cl"
  title: string; // e.g., "APR FC vs Al Ahly"
  competition: string; // e.g., "CAF Champions League"
  sport: "Football" | "Basketball" | "Volleyball";
  homeTeam: string; // e.g., "APR FC"
  awayTeam: string; // e.g., "Al Ahly"
  kickoffIso: string; // e.g., "2026-11-25T21:00:00+02:00" (CAT timezone!)
  endIso?: string; // Approx 2 hours after kickoff
}

interface MatchBroadcastJsonLdProps {
  match: MatchFixture;
  confirmedVenues: MatchdayVenue[];
  baseUrl?: string;
}

export function MatchBroadcastJsonLd({
  match,
  confirmedVenues,
  baseUrl = "https://ruhago.xyz",
}: MatchBroadcastJsonLdProps) {
  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "SportsEvent",
    "@id": `${baseUrl}/matches/${match.slug}#event`,
    name: `${match.title} — Where to Watch Live in Kigali`,
    description: `Watch ${match.title} (${match.competition}) live in Kigali. Compare ${confirmedVenues.length} verified sports bars, lounges, and fan houses showing the match with commentary audio and backup power.`,
    sport: match.sport,
    startDate: match.kickoffIso,
    endDate: match.endIso || match.kickoffIso,
    eventStatus: "https://schema.org",
    eventAttendanceMode: "https://schema.org",
    homeTeam: {
      "@type": "SportsTeam",
      name: match.homeTeam,
    },
    awayTeam: {
      "@type": "SportsTeam",
      name: match.awayTeam,
    },
    organizer: {
      "@type": "Organization",
      name: match.competition,
    },
    // Links the match to physical Kigali bars hosting live watch parties
    location: confirmedVenues.map((venue) => ({
      "@type": "SportsActivityLocation",
      name: venue.name,
      url: `${baseUrl}/places/${venue.slug}`,
      address: {
        "@type": "PostalAddress",
        streetAddress: venue.location.address,
        addressLocality: venue.location.neighborhood,
        addressRegion: "Kigali",
        addressCountry: "RW",
      },
      geo: {
        "@type": "GeoCoordinates",
        latitude: venue.location.coordinates.lat,
        longitude: venue.location.coordinates.lng,
      },
    })),
    // Tells Google & AI engines this is a live public screening/broadcast
    subEvent: confirmedVenues.map((venue) => ({
      "@type": "BroadcastEvent",
      name: `Live Screening: ${match.title} at ${venue.name}`,
      isLiveBroadcast: true,
      startDate: match.kickoffIso,
      videoFormat: "HD",
      broadcastOfEvent: {
        "@type": "SportsEvent",
        name: match.title,
      },
      location: {
        "@type": "Place",
        name: venue.name,
        address: `${venue.location.address}, Kigali, Rwanda`,
      },
    })),
  };

  return (
    <script
      type="application/ld+json"
      dangerouslySetInnerHTML={{
        __html: JSON.stringify(jsonLd).replace(/</g, "\\u003c"),
      }}
    />
  );
}
```
